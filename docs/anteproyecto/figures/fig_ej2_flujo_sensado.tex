\begin{tikzpicture}[
  >=stealth, thick,
  font=\small,
  node distance=1.2cm,
  startstop/.style={draw=red!70!black, fill=red!10, rectangle, rounded corners=4pt, minimum height=0.9cm, minimum width=3.2cm, align=center, font=\small\bfseries},
  process/.style={draw=blue!70!black, fill=blue!10, rectangle, rounded corners=4pt, minimum height=1.0cm, minimum width=3.4cm, align=center},
  decision/.style={draw=yellow!60!black, fill=yellow!20, diamond, aspect=2, inner sep=3pt, align=center, font=\small}
]

  \node[startstop] (start) {Inicio de Muestreo};
  \node[process, below=of start] (read) {Toma de Caída de\\Tensión en Shunt ($V_{shunt}$)};
  \node[process, below=of read] (amp) {Acondicionamiento Diferencial\\y Filtrado Analógico};
  \node[decision, below=1.3cm of amp] (check) {¿$V_{ana} > V_{ref}$?\\{\footnotesize (¿Sobrecorriente / Tope?)}};
  
  \node[process, right=1.6cm of check] (adc) {Conversión ADC\\y Monitoreo Continuo};
  \node[process, below=1.3cm of check] (interrupt) {Disparo de Comparador e\\Interrupción Prioritaria};
  \node[startstop, below=of interrupt] (stop) {Inhibición de Movimiento\\(Parada de Seguridad)};

  % Conexiones de flujo
  \draw[->] (start) -- (read);
  \draw[->] (read) -- (amp);
  \draw[->] (amp) -- (check);
  
  % Rama Sí
  \draw[->] (check) -- node[right, font=\footnotesize\bfseries] {Sí} (interrupt);
  \draw[->] (interrupt) -- (stop);
  
  % Rama No (Lazo de retorno limpio hacia el inicio de la lectura)
  \draw[->] (check) -- node[above, font=\footnotesize\bfseries] {No} (adc);
  \draw[->] (adc.north) |- (read.east);
